Der Eintrag EPSG:3907 steht für das Koordinatenreferenzsystem “MGI 1901 / Balkans zone 5”. Es handelt sich dabei um ein historisches, projiziertes Koordinatensystem (eingeführt 1924), das für den Westen Kroatiens verwendet wurde und in Metern (Rechtswert/Hochwert) statt in Grad gemessen wird. Inzwischen ist es in Kroatien eigentlich durch HTRS96 / Croatia TM (EPSG:3765) abgelöst worden. [KI]

from pathlib import Path
import duckdb
import geopandas as gpd
import pandas as pd


def convert_latest_geojson_to_geoparquet(data_raw: Path) -> Path:
    geojson_files = sorted(data_raw.glob("*_geoportalnipp_rkd.geojson"))

    if not geojson_files:
        raise FileNotFoundError(
            "Keine passenden GeoJSON-Dateien im Verzeichnis gefunden."
        )

    latest_geojson = geojson_files[-1]
    parquet_output = latest_geojson.with_suffix(".parquet")
    print(f"Aktuellste Datei: {latest_geojson.name}")

    # GeoJSON mit GeoPandas laden
    gdf = gpd.read_file(latest_geojson)

    if gdf.crs is None:
        gdf.set_crs(epsg=3907, inplace=True)

    gdf.to_parquet(parquet_output)

    print("Konvertierung zu GeoParquet erfolgreich abgeschlossen.")
    print(f"Parquet-Datei: {parquet_output.name}\n")
    return parquet_output


def extract_metadata(df: pd.DataFrame) -> pd.DataFrame:
    total_rows = len(df)
    metadata = []
    for idx, col in enumerate(df.columns):
        dtype = df[col].dtype
        num_nan = df[col].isna().sum()
        num_unique = df[col].nunique()
        valid_rows = total_rows - num_nan
        pct_nan = (num_nan / total_rows) * 100
        pct_unique = (num_unique / valid_rows * 100) if valid_rows > 0 else 0.0

        max_str_len = None
        starts_with_whitespace = 0
        contains_newline = 0

        if pd.api.types.is_string_dtype(dtype) or dtype == "object":
            col_as_str = df[col].astype(str)
            if num_nan < total_rows:
                max_str_len = df[col].str.len().max()
            starts_with_whitespace = col_as_str.str.match(r'^\s').sum()
            contains_newline = col_as_str.str.contains(r"[\r\n]").sum()

        metadata.append(
            {
                "index": idx,
                "column_name": col,
                "type": str(dtype),
                "nan": num_nan,
                "nan_pct": f"{pct_nan:.1f}%",
                "uniques": num_unique,
                "uniques_pct": f"{pct_unique:.1f}%",
                "max_len": (
                    int(max_str_len) if pd.notna(max_str_len) else "-"
                ),
                "starts_whitespace": starts_with_whitespace,
                "contains_newline": contains_newline,
            }
        )
    return pd.DataFrame(metadata).set_index("index")
