Erstellen eines Loggings, damit alle Änderungen am Original-Datensatz in einer Log-Datei gespeichert werden.

def setup_dynamic_logging(base_filename: str) -> logging.Logger:
    """Configures and retrieves a file-specific logger instance.

    Creates a logger that writes log entries simultaneously to a dedicated log
    file inside the 'data/interim' directory and to the standard console output.
    If a logger with the given name already exists and has active handlers, it
    is returned as-is to prevent duplicate log messages during interactive
    execution environments like Quarto or Jupyter notebooks.

    Args:
        base_filename: The base name of the file being processed, typically the
        stem of the input file path without extensions.

    Returns:
        A logging.Logger instance configured with file and stream handlers
        using a standardized timestamp and log level format.
    """
    logger = logging.getLogger(base_filename)
    logger.setLevel(logging.INFO)
    logger.propagate = False
    if logger.handlers:
        return logger

    log_file_path = DATA_INTERIM / f"cleaned_{base_filename}.log"

    formatter = logging.Formatter("%(asctime)s [%(levelname)s] %(message)s")

    file_handler = logging.FileHandler(log_file_path, encoding="utf-8")
    file_handler.setFormatter(formatter)
    logger.addHandler(file_handler)

    stream_handler = logging.StreamHandler()
    stream_handler.setFormatter(formatter)
    logger.addHandler(stream_handler)

    return logger

Grundbereinigung der Strings

Wie bereits festgestellt, müssen führende Leerzeichen (und ggf. am Ende) und Excel-Artefakte (Zeilenumbrüche) entfernt werden. Hinzu kommt, dass fehlende Werte in den Daten mit dem String “-” ersetzt wurden.

NoteModul: text_cleaning.py 
import geopandas as gpd


def clean_text_columns(gdf: gpd.GeoDataFrame, logger) -> gpd.GeoDataFrame:
    """Cleans text columns within a GeoDataFrame and logs the modifications.

    Iterates through all columns of type 'object' or 'string', removes carriage
    returns, strips leading/trailing whitespaces, and converts specific empty 
    or placeholder values into Python None objects. Each modification is 
    tracked and summarized using the provided logger instance.

    Args:
        gdf: The input GeoDataFrame containing spatial and attribute data.
        logger: A logging.Logger instance configured to record the cleaning 
            process details and summary statistics.

    Returns:
        A new GeoDataFrame instance containing the cleaned and normalized 
        text fields, while maintaining the original geometry and CRS.
    """
    logger.info("Schritt 1: Allgemeiner Text-Clean (Strip, Linebreaks, Platzhalter)...")
    gdf = gdf.copy()
    text_columns = gdf.select_dtypes(include=["object", "string"]).columns
    total_updates = 0

    for col in text_columns:
        orig_series = gdf[col].astype(str)
        cleaned_series = (
            orig_series.str.replace(r"\n\r", " ", regex=True)
            .str.strip()
            .str.replace(r"^-$", "", regex=True)
            .replace(["", "nan", "None"], None)
        )
        changed_mask = orig_series != cleaned_series.astype(str)
        num_changed = changed_mask.sum()

        if num_changed > 0:
            gdf[col] = cleaned_series
            logger.info(f"Spalte '{col}': {num_changed}")
            total_updates += num_changed

    logger.info(f"Allgemeiner Text-Clean beendet. {total_updates} Änderungen.")
    return gdf

import geopandas as gpd


def clean_column_grad(gdf: gpd.GeoDataFrame, logger) -> gpd.GeoDataFrame:
    """Standardizes the 'grad' column to Title Case and splits by delimiter."""
    logger.info("Schritt 2: Bereinige Spalte 'grad'...")

import geopandas as gpd


def drop_column_adresa(
    gdf: gpd.GeoDataFrame, logger
) -> gpd.GeoDataFrame:
    """Removes the 'adresa' column completely from the GeoDataFrame.

    Args:
        gdf: The input GeoDataFrame.
        logger: A logging.Logger instance to record the removal.

    Returns:
        A new GeoDataFrame without the 'adresa' column.
    """
    logger.info("Schritt 3: Entferne Spalte 'adresa' vollständig...")
    gdf = gdf.copy()

    if "adresa" in gdf.columns:
        gdf = gdf.drop(columns=["adresa"])
        logger.info("Spalte 'adresa' wurde erfolgreich gelöscht.")
    else:
        logger.warning("Spalte 'adresa' existiert bereits nicht mehr.")

    return gdf

def rename_vrsta_categories(
    gdf: gpd.GeoDataFrame, logger
) -> gpd.GeoDataFrame:
    """Translates 'vrsta' column values to match the official ministry register.

    Applies a fixed mapping to standardize all four unique categories in the
    'vrsta' column to their official terms from the Ministry of Culture.

    Args:
        gdf: The input GeoDataFrame containing the 'vrsta' column.
        logger: A logging.Logger instance to record processing changes.

    Returns:
        A new GeoDataFrame with updated 'vrsta' category names.
    """
    logger.info(
        "Schritt: Passe Spalte 'vrsta' an das offizielle Webregister an..."
    )
    gdf = gdf.copy()

    if "vrsta" in gdf.columns:
        vrsta_mapping = {
            "Pojedinačna kulturna dobra": "Nepokretna pojedinačna",
            "Kulturnopovijesne cjeline": "Kulturnopovijesna cjelina",
            "Kulturni krajolici": "Kulturni krajolik",
            "Arheološka kulturna dobra": "Arheologija",
        }

        orig_series = gdf["vrsta"].astype(str)
        cleaned_series = orig_series.map(vrsta_mapping).fillna(orig_series)

        changed_mask = orig_series != cleaned_series
        num_changed = changed_mask.sum()

        gdf["vrsta"] = cleaned_series

        logger.info(
            f"Spalte 'vrsta': {num_changed} Datensätze auf Register-Namen aktualisiert."
        )
    else:
        logger.warning("Spalte 'vrsta' nicht im GeoDataFrame gefunden!")

    return gdf

def check_klasifikacija_against_register(
    gdf: gpd.GeoDataFrame, logger: logging.Logger
) -> None:
    """Checks the 'klasifikacija' column against the official web register.

    Prints and logs any unexpected values found in the column to justify
    subsequent data cleaning steps.

    Args:
        gdf: The GeoDataFrame to analyze.
        logger: A logging.Logger instance to record the results.
    """
    logger.info("Überprüfe Spalte 'klasifikacija' gegen das offizielle Webregister...")

    official_values = [
        "javne građevine", "stambene građevine", "stambeno-poslovne građevine",
        "stambeno-gospodarske građevine", "sakralne građevine", "sakralni kompleksi",
        "sakralno obilježje u prostoru", "gospodarske građevine", "industrijske građevine",
        "poslovne građevine", "vojne i obrambene građevine", "memorijalne građevine",
        "memorijalna obilježja i mjesta", "komunalna i tehnička građevina", "javna plastika",
        "urbana oprema", "memorijalna cjelina", "ruralna cjelina", "urbana cjelina",
        "gospodarska i industrijska cjelina", "uređene zelene površine", "vojna i obrambena cjelina",
        "planirani krajolik", "organski krajolik", "asocijativni krajolik",
        "kopnena arheološka zona/nalazište", "podvodna arheološka zona/nalazište",
        "kopneno/podvodno arheološka zona/nalazište"
    ]

    if "klasifikacija" not in gdf.columns:
        logger.warning("Spalte 'klasifikacija' nicht im Datensatz vorhanden.")
        return

    # Finde alle Werte, die NICHT in der offiziellen Liste sind (ohne NaN)
    invalid_mask = ~gdf["klasifikacija"].isin(official_values) & gdf["klasifikacija"].notna()
    invalid_counts = gdf.loc[invalid_mask, "klasifikacija"].value_counts()

    # Ausgabe steuern
    print("\nINTERVALL-CHECK: ABWEICHUNGEN VOM WEBREGISTER")
    if not invalid_counts.empty:
        msg = f"Gefundene Werte, die NICHT im offiziellen Register existieren:\n{invalid_counts.to_string()}"
        print(msg)
        logger.warning(f"Abweichungen in 'klasifikacija' gefunden:\n{invalid_counts.to_string()}")
    else:
        msg = "Perfekt! Alle Werte in 'klasifikacija' entsprechen dem offiziellen Register."
        print(msg)
        logger.info(msg)

import geopandas as gpd


def replace_value_in_column(
    gdf: gpd.GeoDataFrame,
    column: str,
    old_value: str,
    new_value: str,
    logger,
) -> gpd.GeoDataFrame:
    """Replaces a specific value in a column and logs the number of changes.

    Args:
        gdf: The input GeoDataFrame.
        column: Name of the target column.
        old_value: The exact string value to be replaced.
        new_value: The new string value to insert.
        logger: Logger instance for tracking the update.

    Returns:
        A new GeoDataFrame with the value replaced.
    """
    gdf = gdf.copy()

    if column in gdf.columns:
        # Maske für den zu ändernden Wert
        mask = gdf[column] == old_value
        num_changed = mask.sum()

        if num_changed > 0:
            gdf.loc[mask, column] = new_value
            logger.info(
                f"Schritt: Wert in '{column}' angepasst. "
                f"'{old_value}' -> '{new_value}' ({num_changed} Feld(er) korrigiert)."
            )
        else:
            logger.info(
                f"Schritt: Keine Anpassung in '{column}'. "
                f"Wert '{old_value}' wurde nicht gefunden."
            )
    else:
        logger.warning(f"Schritt fehlgeschlagen: Spalte '{column}' existiert nicht.")

    return gdf
